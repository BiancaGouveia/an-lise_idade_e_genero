function verificando() { var imagem = window.document.getElementById('imagem') var data = new Date() var anoAtual = data.getFullYear() var anoRegistrado = window.document.getElementById('txtAno').value if (anoRegistrado.length == 0 || anoRegistrado > anoAtual) { window.alert('[situação: ERRO] Informe seus dados corretamente e tente novamente') } else { var genero = "" var formularioSex = window.document.getElementsByName('txtSexo') var res = window.document.getElementById('res') var idade = anoAtual - anoRegistrado if (formularioSex[0].checked) { genero = 'mulher' if (idade >= 0 && idade < 13) { imagem.src = 'menina.bebe.png' } else if (idade < 25) { imagem.src = 'mulher.jovem.jpg' } else if (idade < 60) { imagem.src = 'mulher.png' } else { imagem.src = 'mulher.idosa.jpg' } } else if (formularioSex[1].checked) { genero = 'homem' if (idade >= 0 && idade < 13) { imagem.src = 'menino.bebe.jpg' } else if (idade < 25) { imagem.src = 'homem.jovem.jpg' } else if (idade < 60) { imagem.src = 'homem.png' } else { imagem.src = 'homem.idoso.jpg' } } res.innerHTML = `Você é ${genero} e tem ${idade} anos de idade` } }
